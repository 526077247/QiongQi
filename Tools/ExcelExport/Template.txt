export class (ConfigName)Category{

    private static _instance: (ConfigName)Category;
    public static get instance():(ConfigName)Category{
        if((ConfigName)Category._instance != null) return (ConfigName)Category._instance
        JsonHelper.registerClass((ConfigName),'(ConfigName)');
        JsonHelper.registerClass((ConfigName)Category,'(ConfigName)Category');
        const category = JsonHelper.deserialize((ConfigName)Category, Data) as (ConfigName)Category;
        category.endInit();
        (ConfigName)Category._instance = category;
    }

    private dict = new Map<number, (ConfigName)>();

    private list:(ConfigName)[] = [];

    public endInit()
    {
        for(let i =0 ;i<this.list.length;i++)
        {
            const config:(ConfigName) = this.list[i];

            this.dict.set(config.id, config);
        }            
    }
    
    public get(id: number): (ConfigName)
    {
        let item: (ConfigName) = this.dict.get(id);
        
        if (item == null)
        {
            Log.error("配置找不到，配置表名: (ConfigName)，配置id: "+id);
        }

        return item;
    }

    public contain(id: number): boolean
    {
        return this.dict.has(id);
    }

    public getAll(): Map<number, (ConfigName)>
    {
        return this.dict;
    }

    public getAllList(): (ConfigName)[]
    {
        return this.list;
    }
}
